#!/usr/bin/env bash
set -euo pipefail

: "${CUR_SCRIPT:=$(basename "${BASH_SOURCE[0]:-${0:-pm-rollback-gcp}}")}"
export CUR_SCRIPT

: "${PORTMASON_SHARE:=$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd -P)}"
# shellcheck disable=SC1090
. "${PORTMASON_SHARE}/pm-helpers" --read-only

prompt_for_index() {
    local output_var="${1:?output variable required}"
    local prompt="${2:?prompt required}"
    local max="${3:?maximum required}"
    local response

    while true; do
        if ! read -r -p "${prompt} [1-${max}, q to quit]: " response; then
            die "No selection received" 2
        fi

        case "$response" in
            q|Q)
                log INFO "Rollback cancelled"
                exit 0
                ;;
            ''|*[!0-9]*)
                printf 'Enter a number from 1 to %s, or q to quit.\n' "$max" >&2
                ;;
            *)
                if (( response >= 1 && response <= max )); then
                    printf -v "$output_var" '%s' "$response"
                    return 0
                fi
                printf 'Enter a number from 1 to %s, or q to quit.\n' "$max" >&2
                ;;
        esac
    done
}

main() {
    local services_json
    local service_choice=1
    local selected_service=""
    local selected_region=""
    local service_name service_region
    local revisions_json service_json verified_service_json
    local revision_choice selected_revision selected_ready
    local revision_name revision_created revision_ready
    local confirmation traffic_percent
    local active_marker active_revision
    local -a service_rows=()
    local -a revision_rows=()
    local -a active_revisions=()

    have gcloud || die "gcloud is required for GCP rollback" 2
    have jq || die "jq is required for GCP rollback" 2
    [[ -n "${DEPLOYMENT_ID:-}" ]] || die "DEPLOYMENT_ID is required" 2

    log INFO "Discovering Cloud Run services" "deployment_id=$DEPLOYMENT_ID"
    if ! services_json="$(
        gcloud run services list \
            --project "$DEPLOYMENT_ID" \
            --platform managed \
            --format json
    )"; then
        die "Unable to list Cloud Run services for deployment $DEPLOYMENT_ID" 3
    fi

    mapfile -t service_rows < <(
        jq -r '
            .[]
            | [
                .metadata.name,
                (
                    .metadata.labels["cloud.googleapis.com/location"]
                    // .metadata.annotations["run.googleapis.com/region"]
                    // ""
                )
            ]
            | @tsv
        ' <<<"$services_json"
    )

    (( ${#service_rows[@]} > 0 )) \
        || die "No Cloud Run services found in deployment $DEPLOYMENT_ID. Cloud Run jobs cannot be rolled back by revision." 3

    if (( ${#service_rows[@]} == 1 )); then
        IFS=$'\t' read -r selected_service selected_region <<<"${service_rows[0]}"
        log INFO "Using the only Cloud Run service" \
            "service=$selected_service" \
            "region=${selected_region:-unknown}"
    else
        printf '\nCloud Run services in %s:\n' "$DEPLOYMENT_ID"
        for ((service_choice = 0; service_choice < ${#service_rows[@]}; service_choice++)); do
            IFS=$'\t' read -r service_name service_region <<<"${service_rows[$service_choice]}"
            printf '  %d) %s (%s)\n' \
                "$((service_choice + 1))" \
                "$service_name" \
                "${service_region:-region unavailable}"
        done

        prompt_for_index service_choice "Select a service" "${#service_rows[@]}"
        IFS=$'\t' read -r selected_service selected_region \
            <<<"${service_rows[$((service_choice - 1))]}"
    fi

    [[ -n "$selected_region" ]] \
        || die "Cloud Run region is unavailable for service $selected_service" 3

    if ! service_json="$(
        gcloud run services describe "$selected_service" \
            --project "$DEPLOYMENT_ID" \
            --region "$selected_region" \
            --platform managed \
            --format json
    )"; then
        die "Unable to describe Cloud Run service $selected_service" 3
    fi

    mapfile -t active_revisions < <(
        jq -r '
            .status.traffic[]?
            | select((.percent // 0) > 0)
            | .revisionName // empty
        ' <<<"$service_json"
    )

    if ! revisions_json="$(
        gcloud run revisions list \
            --project "$DEPLOYMENT_ID" \
            --region "$selected_region" \
            --service "$selected_service" \
            --platform managed \
            --sort-by '~metadata.creationTimestamp' \
            --limit 10 \
            --format json
    )"; then
        die "Unable to list revisions for Cloud Run service $selected_service" 3
    fi

    mapfile -t revision_rows < <(
        jq -r '
            .[]
            | [
                .metadata.name,
                .metadata.creationTimestamp,
                (
                    [
                        .status.conditions[]?
                        | select(.type == "Ready")
                        | .status
                    ][0] // "Unknown"
                )
            ]
            | @tsv
        ' <<<"$revisions_json"
    )

    (( ${#revision_rows[@]} > 0 )) \
        || die "No Cloud Run revisions found for service $selected_service" 3

    printf '\nFive newest revisions for %s (%s):\n' \
        "$selected_service" "$selected_region"
    for ((revision_choice = 0; revision_choice < ${#revision_rows[@]}; revision_choice++)); do
        IFS=$'\t' read -r revision_name revision_created revision_ready \
            <<<"${revision_rows[$revision_choice]}"

        active_marker=""
        for active_revision in "${active_revisions[@]}"; do
            if [[ "$active_revision" == "$revision_name" ]]; then
                active_marker=" [ACTIVE]"
                break
            fi
        done

        printf '  %d) %s  %s  Ready=%s%s\n' \
            "$((revision_choice + 1))" \
            "$revision_name" \
            "$revision_created" \
            "$revision_ready" \
            "$active_marker"
    done

    prompt_for_index revision_choice "Select a revision" "${#revision_rows[@]}"
    IFS=$'\t' read -r selected_revision revision_created selected_ready \
        <<<"${revision_rows[$((revision_choice - 1))]}"

    [[ "$selected_ready" == "True" ]] \
        || die "Revision $selected_revision is not ready and cannot receive traffic" 4

    printf '\nRoute 100%% of %s traffic to %s?\n' \
        "$selected_service" "$selected_revision"
    if ! read -r -p "Proceed? [y/N]: " confirmation; then
        die "No confirmation received" 2
    fi

    case "$confirmation" in
        y|Y|yes|YES)
            ;;
        *)
            log INFO "Rollback cancelled"
            return 0
            ;;
    esac

    gcloud run services update-traffic "$selected_service" \
        --project "$DEPLOYMENT_ID" \
        --region "$selected_region" \
        --platform managed \
        --to-revisions="${selected_revision}=100"

    if ! verified_service_json="$(
        gcloud run services describe "$selected_service" \
            --project "$DEPLOYMENT_ID" \
            --region "$selected_region" \
            --platform managed \
            --format json
    )"; then
        die "Traffic was updated, but verification failed for service $selected_service" 5
    fi

    traffic_percent="$(
        jq -r --arg revision "$selected_revision" '
            [
                .status.traffic[]?
                | select(.revisionName == $revision)
                | .percent
            ][0] // 0
        ' <<<"$verified_service_json"
    )"

    [[ "$traffic_percent" == "100" ]] \
        || die "Rollback verification failed: revision $selected_revision has $traffic_percent% traffic" 5

    log INFO "Cloud Run rollback verified" \
        "deployment_id=$DEPLOYMENT_ID" \
        "service=$selected_service" \
        "region=$selected_region" \
        "revision=$selected_revision" \
        "traffic_percent=$traffic_percent"
}

main "$@"
